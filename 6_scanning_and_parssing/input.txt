# Define a function to find the first k prime numbers
def find_primes():
    k = input()
    num = 2
    count = 0

    while count < k:
        isPrime = True
        i = 2

        while i < num:
            if num % i == 0:
                isPrime = False
                break
            i = i + 1

        if isPrime:
            print(num)
            count = count + 1

        num = num + 1

# Call the function
find_primes()
